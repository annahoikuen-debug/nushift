```python
def _collect_highs_info(prob: Any, stats: dict) -> None:
    """HiGHS 実行後のギャップ・双境界・ノード数を stats に記録する。

    CBC など ``solverModel`` を持たないソルバーでは何もしない。
    ``getInfo()`` は solve 後のみ有効なので、呼び出しは ``_run_solver`` 内
    （``ctx.prob.solve`` の直後）に限る。
    """
    model = getattr(prob, "solverModel", None)
    if model is None:
        return
    try:
        info = model.getInfo()
    except Exception:
        return
    try:
        stats["mip_gap"] = float(info.mip_gap)
    except (TypeError, ValueError):
        stats["mip_gap"] = None
    try:
        stats["mip_dual_bound"] = float(info.mip_dual_bound)
    except (TypeError, ValueError):
        stats["mip_dual_bound"] = None
    try:
        stats["mip_node_count"] = int(info.mip_node_count)
    except (TypeError, ValueError):
        stats["mip_node_count"] = None
```

`Any` は solver.py の 40行目付近で既に import 済み（`from typing import Any`）。

### ステップ13: `highs_start.py` を新規作成（`HiGHSWithStart` クラス）

**対象**: `src/shiftai/highs_start.py`（新規）

**作業**: 次の内容で新規ファイルを作成する（検証済みコードをそのまま使用）:

```python
"""HiGHS に初期解（MIP スタート）を渡すソルバーサブクラス。

``pulp.HiGHS`` は ``actualSolve`` 内で ``callSolver`` を呼ぶ。
``callSolver`` の直前は ``lp.solverModel``（``highspy.Highs`` インスタンス）
が構築済みで、各変数には ``var.index`` と ``var.varValue`` が設定されている。
このタイミングで ``setSolution`` を呼ぶと HiGHS が初期解として採用する。

実測（highspy 1.16.0 / pulp 2.9.0）: ``setSolution(len, indices, values)`` は
``HighsStatus.kOk`` を返し、小問題では 0 ノードで最適解に到達した。
``setInitialValue`` は ``varValue`` に格納されるため、``apply_warm_start``
が設定した値がそのまま初期解になる。
"""
from __future__ import annotations

import pulp


class HiGHSWithStart(pulp.HiGHS):
    """``callSolver`` 直前に ``varValue`` を MIP スタートとして HiGHS へ渡す。"""

    def callSolver(self, lp: pulp.LpProblem) -> None:
        try:
            idx: list[int] = []
            val: list[float] = []
            for var in lp.variables():
                v = var.varValue
                if v is not None:
                    idx.append(var.index)
                    val.append(float(v))
            if idx:
                lp.solverModel.setSolution(len(idx), idx, val)
        except Exception:
            # 初期解の投入は最適化。失敗しても通常通り solve する。
            pass
        super().callSolver(lp)
```

### ステップ24: `weekly.py` を新規作成（`solve_weekly` 関数）

**目的**: 週次（7日）を `solve_monthly_hierarchical` で解く薄いラッパーを提供する。既存の `solve_shift` との互換シグネチャ + `engine`/`parallel_workers` 引数を持つ。

**対象**: `src/shiftai/weekly.py`（新規）

**作業**: 次の内容で新規ファイルを作成する:

```python
"""週次シフト求解の薄いラッパー。

``solve_shift`` と同シグネチャ + ``engine`` / ``parallel_workers`` を持つ。
``engine="hierarchical"``（既定）の場合は ``solve_monthly_hierarchical`` に
日単位チャンク（chunk_size=1）で委譲する。高度機能（pairing_rules / classrooms /
require_leader / require_opening_closing_skill / drop_groups / relaxation != STRICT）
を使用する場合、または ``engine="legacy"`` の場合は ``solve_shift`` に
フォールバックする（既存の高度機能を維持するため）。
"""
from __future__ import annotations

from collections.abc import Mapping, Sequence
from datetime import date

from shiftai import config
from shiftai.domain import (
    CellState,
    ChildPlan,
    FacilitySettings,
    ObjectiveWeights,
    PairingRule,
    RequirementTable,
    RelaxLevel,
    ShiftPattern,
    SolveResult,
    StaffMember,
    StaffPreferences,
    StaffingStandard,
    ClassRoom,
)


def solve_weekly(
    children: Sequence[ChildPlan],
    staff: Sequence[StaffMember],
    requirements: RequirementTable,
    preferences: Mapping[str, StaffPreferences] | None = None,
    weights: ObjectiveWeights | None = None,
    *,
    fixed_assignments: Mapping[tuple[str, date, str], CellState] | None = None,
    settings: FacilitySettings | None = None,
    time_limit_sec: int = config.DEFAULT_TIME_LIMIT_SEC,
    msg: bool = False,
    standard: StaffingStandard | None = None,
    patterns: Sequence[ShiftPattern] | None = None,
    relaxation: int = int(RelaxLevel.STRICT),
    drop_groups: frozenset[str] = frozenset(),
    solver_name: str = config.DEFAULT_SOLVER,
    warm_start: bool = True,
    warm_start_assignments: Mapping[tuple[str, date, str], CellState] | None = None,
    pattern_mode: bool = True,
    pairing_rules: Sequence[PairingRule] | None = None,
    classrooms: Sequence[ClassRoom] | None = None,
    require_leader: bool = False,
    require_opening_closing_skill: bool = False,
    engine: str = config.WEEKLY_ENGINE_DEFAULT,
    parallel_workers: int = config.WEEKLY_PARALLEL_WORKERS,
) -> SolveResult:
    """週次シフトを求解する。

    ``engine="hierarchical"``（既定）: ``solve_monthly_hierarchical`` に委譲。
    ``engine="legacy"`` または高度機能使用時: ``solve_shift`` にフォールバック。
    """
    # 高度機能または明示指定時は solve_shift にフォールバック
    has_advanced = bool(
        pairing_rules
        or classrooms
        or require_leader
        or require_opening_closing_skill
        or drop_groups
        or (relaxation != int(RelaxLevel.STRICT))
    )
    if engine == "legacy" or has_advanced:
        from shiftai.solver import solve_shift
        return solve_shift(
            children,
            staff,
            requirements,
            preferences,
            weights,
            fixed_assignments=fixed_assignments,
            settings=settings,
            time_limit_sec=time_limit_sec,
            msg=msg,
            standard=standard,
            patterns=patterns,
            relaxation=relaxation,
            drop_groups=drop_groups,
            solver_name=solver_name,
            warm_start=warm_start,
            warm_start_assignments=warm_start_assignments,
            pattern_mode=pattern_mode,
            pairing_rules=pairing_rules,
            classrooms=classrooms,
            require_leader=require_leader,
            require_opening_closing_skill=require_opening_closing_skill,
        )

    # hierarchical エンジン: 月次階層エンジンに日単位チャンクで委譲
    from shiftai.monthly import MonthlySolveContext, solve_monthly_hierarchical

    all_days = requirements.all_days()
    contracts_map = {s.staff_id: s.contract for s in staff if s.contract}
    ctx = MonthlySolveContext(
        target_month=f"{all_days[0].year:04d}-{all_days[0].month:02d}" if all_days else "",
        all_days=all_days,
        staff=staff,
        contracts=contracts_map,
        preferences=preferences,
        patterns=patterns or (),
        requirements=requirements,
        facility_settings=settings or FacilitySettings(),
        fixed_assignments=fixed_assignments or {},
        prior_duty_history={},
        monthly_holiday_target=8,
        children=children,
        standard=standard,
        weights=weights,
    )
    return solve_monthly_hierarchical(
        ctx,
        time_limit_sec=time_limit_sec,
        solver_name=solver_name,
        parallel_workers=parallel_workers,
    )
```

K3: stats["solver"] ≠ greedy（MILP で解けた）
K4: MIPギャップ ≤ 3% 表示（7日×28名）

使用例:
    cd /home/annahoikuen/nushift/nushift
    PYTHONPATH=src python3 scripts/bench_kpi.py
"""
from __future__ import annotations

import sys
import time
from datetime import date, timedelta

from shiftai import local_rules, sample_data
from shiftai.config import DEFAULT_DAY_OPEN, DEFAULT_DAY_CLOSE, DEFAULT_GRANULARITY_MIN
from shiftai.domain import FacilitySettings, ViolationSeverity
from shiftai.standards import build_requirements
from shiftai.solver import solve_shift

STANDARD = local_rules.get_standard("全国基準（厚労省）")


def run_bench(days_count: int, time_limit: int = 60) -> dict:
    """指定日数のベンチを実行し、結果を返す。"""
    days = [date(2026, 11, 16) + timedelta(days=i) for i in range(days_count)]
    children, staff, prefs = sample_data.make_dataset(days, seed=42)
    settings = FacilitySettings()
    table = build_requirements(
        children, days, STANDARD,
        day_open=DEFAULT_DAY_OPEN, day_close=DEFAULT_DAY_CLOSE,
        granularity_min=DEFAULT_GRANULARITY_MIN,
    )

    t0 = time.perf_counter()
    res = solve_shift(
        children, staff, table, preferences=prefs, settings=settings,
        time_limit_sec=time_limit, standard=STANDARD,
    )
    wall = time.perf_counter() - t0

    s = res.stats
    blockers = [v for v in res.violations if v.severity == ViolationSeverity.BLOCKER]
    k1 = len(blockers) == 0
    k2 = wall <= 60
    k3 = s.get("solver") != "greedy" and s.get("fallback") != "greedy"
    gap = s.get("mip_gap")
    k4 = gap is not None and float(gap) <= 0.03

    return {
        "days": days_count,
        "wall": wall,
        "solver": s.get("solver"),
        "status": res.status.value,
        "violations": len(res.violations),
        "blockers": len(blockers),
        "mip_gap": gap,
        "K1": "✅" if k1 else "❌",
        "K2": "✅" if k2 else "❌",
        "K3": "✅" if k3 else "❌",
        "K4": "✅" if k4 else "❌",
    }


def main():
    print("=" * 60)
    print("KPI 測定: 7日×28名（週次）")
    print("=" * 60)
    r7 = run_bench(7, time_limit=60)
    print(f"  wall={r7['wall']:.1f}s solver={r7['solver']} status={r7['status']}")
    print(f"  violations={r7['violations']} blockers={r7['blockers']} mip_gap={r7['mip_gap']}")
    print(f"  K1(BLOCKER 0): {r7['K1']}  K2(wall≤60s): {r7['K2']}")
    print(f"  K3(solver≠greedy): {r7['K3']}  K4(gap≤3%): {r7['K4']}")

    print()
    print("=" * 60)
    print("KPI 測定: 30日×28名（月次）")
    print("=" * 60)
    r30 = run_bench(30, time_limit=120)
    print(f"  wall={r30['wall']:.1f}s solver={r30['solver']} status={r30['status']}")
    print(f"  violations={r30['violations']} blockers={r30['blockers']} mip_gap={r30['mip_gap']}")
    print(f"  K1(BLOCKER 0): {r30['K1']}  K2(wall≤60s): {r30['K2']}")
    print(f"  K3(solver≠greedy): {r30['K3']}  K4(gap≤3%): {r30['K4']}")
    print()

    all_k = all([r7["K1"] == "✅", r7["K2"] == "✅", r7["K3"] == "✅",
                 r30["K1"] == "✅", r30["K2"] == "✅", r30["K3"] == "✅"])
    print("総合判定:", "✅ 全 KPI 達成" if all_k else "⚠ 未達あり（ステップ32〜33を参照）")


if __name__ == "__main__":
    main()
